const express = require('express'); const ytdl = require('@distube/ytdl-core'); const ytpl = require('ytpl'); const ffmpeg = require('fluent-ffmpeg'); const app = express(); app.use(express.json()); // 1. ROTA PARA MAPEAR A PLAYLIST app.get('/api/playlist', async (req, res) => { const playlistUrl = req.query.url; if (!playlistUrl) return res.status(400).json({ error: 'URL ausente' }); try { const playlistId = await ytpl.getPlaylistID(playlistUrl); const playlist = await ytpl(playlistId, { limit: Infinity }); const tracks = playlist.items.map(item => ({ id: item.id, title: item.title, url: item.shortUrl })); res.json({ title: playlist.title, tracks }); } catch (error) { res.status(500).json({ error: 'Erro ao ler a playlist. Verifique se ela é pública.' }); } }); // 2. ROTA DE CONVERSÃO REAL PARA MP3 320 KBPS app.get('/api/download', async (req, res) => { const videoUrl = req.query.url; if (!videoUrl) return res.status(400).send('URL ausente'); try { const info = await ytdl.getInfo(videoUrl); // Limpa o título tirando caracteres que quebram o download no Windows/Linux const title = info.videoDetails.title.replace(/[\\/*?:"<>|]/g, ''); // Configura os cabeçalhos para forçar o navegador a baixar como MP3 res.header('Content-Disposition', `attachment; filename="${title}.mp3"`); res.header('Content-Type', 'audio/mpeg'); // Pega o fluxo de áudio de maior fidelidade do YouTube const audioStream = ytdl(videoUrl, { filter: 'audioonly', quality: 'highestaudio', highWaterMark: 1 << 25 }); // Alimenta o FFmpeg com o fluxo do YouTube, força a conversão para MP3 a 320 kbps e envia para a resposta (res) ffmpeg(audioStream) .toFormat('mp3') .audioBitrate(320) // <--- GARANTE OS 320 KBPS REAIS .on('error', (err) => { console.error('Erro no processamento do FFmpeg:', err.message); }) .pipe(res, { end: true }); } catch (error) { console.error('Erro na requisição:', error); if (!res.headersSent) { res.status(500).send('Erro ao processar áudio.'); } } }); // 3. INTERFACE DO APP (INDEX INTEGRADO) app.get('/', (req, res) => { res.send(` EXNA Downloader - Pure 320kbps

EXNA Downloader

Conversor Dedicado: Baixando listas completas cravadas em 320 kbps.

Aguardando início... 0 / 0
`); }); const PORT = 3000; app.listen(PORT, () => console.log(`🔥 Motor de áudio rodando em http://localhost:${PORT}`));
